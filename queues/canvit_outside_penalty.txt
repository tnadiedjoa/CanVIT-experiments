# CanViT's Writes fine-tuned on rollouts of pretraining length (as probes/write_finetune_short.pt), plus the penalty on

results/c2f_c64_t85_s0_finetune_short_op1.json scripts/ade20k_rollouts.py --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --write-gate probes/write_finetune_short_op1.pt
