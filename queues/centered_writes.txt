# ADE20K: Writes outside the glimpse's crop centered instead of dropped. scripts/write_leakage.py finds that most of
# what leaks over C2F's level 3 is one vector added to every unseen cell; centering removes that vector and keeps the
# rest of the leak, which local Writes from glimpse 1 show to help in C2F's first levels on ADE20K.
c2f_c64_t85_s0_center1 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 1 --outside-writes center
c2f_c64_t85_s0_center21 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 21 --outside-writes center
# Moved up from queues/local_writes.txt (that runner skips them once done): local Writes from level 2, and the second
# seed of local Writes from level 3.
c2f_c64_t85_s0_local5 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 5
c2f_c64_t85_s1_local21 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 21 --seed 1
# Local Writes for small glimpses only (scale ≤ 1/4; larger glimpses write everywhere, as released). For C2F this is
# local Writes from level 2 (c2f_c64_t85_s0_local5). Local Writes from glimpse 1 cost R-IID and F2C several points
# early, when small glimpses leave most of the canvas unseen, and F-IID and C2F ~0.5 point at their large glimpses.
f2c_c64_t21_s0_localmax0.25 --policy fine_to_coarse --num-glimpses 21 --canvas-grid-size 64 --local-writes-from 1 --local-writes-max-scale 0.25
fiid_c64_t85_s0_localmax0.25 --policy full_then_random --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 1 --local-writes-max-scale 0.25
riid_c64_t85_s0_localmax0.25 --policy random --num-glimpses 85 --canvas-grid-size 64 --local-writes-from 1 --local-writes-max-scale 0.25
# The attention sink: outside the crop, canvas cells put ~90% of their Write attention on backbone register 3, whose
# value is not zero. Removing that register's part of every Write, in every cell, needs no viewpoint.
c2f_c64_t85_s0_nosink1 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --dropped-write-token 5
c2f_c64_t85_s0_nosink21 --policy coarse_to_fine --num-glimpses 85 --canvas-grid-size 64 --dropped-write-token 5 --dropped-write-token-from 21
riid_c64_t85_s0_nosink1 --policy random --num-glimpses 85 --canvas-grid-size 64 --dropped-write-token 5
# RFS has no writes outside the crop (its glimpse is the whole scene), yet declines over 85 glimpses; cells inside the
# crop also put ~40% of their attention on the sink register. Does removing it slow the decline?
rfs_c64_t85_nosink1 --policy repeated_full_scene --num-glimpses 85 --canvas-grid-size 64 --dropped-write-token 5
